{} 对象创建与访问

// 对象字面量 let person = { name: "张耀夫", age: 19, grade: "计算机24009" }; // 两种访问方式 person.name; // "." 常用 person["age"]; // "[]" 支持变量/特殊字符 // 修改 / 新增属性 person.age = 20; // 修改 person.email = "1827735290@qq.com"; // 新增 // 删除属性 delete person.grade;
属性名： 值：

for...in / Object.keys / Object.values

// 遍历键名 for (let key in person) { console.log(key, person[key]); } // 静态方法（返回数组） Object.keys(person); // → [所有键名] Object.values(person); // → [所有值] Object.entries(person); // → [[key,val], ...]

JSON.stringify / JSON.parse — 序列化

// 对象 ↔ JSON 字符串 let str = JSON.stringify(person); let obj = JSON.parse(str); // str 可传给服务器，obj 是全新的对象副本
② 字符串（String）
替换文本：

.length / [index] / charAt()

let s = "Hello"; s.length; // 5 s[0]; // "H" s.charAt(4); // "o" s.charCodeAt(0); // 72 (ASCII)
索引：

indexOf / includes / startsWith / endsWith

let s = "Hello JavaScript"; s.indexOf("Java"); // 6（找不到返回 -1） s.includes("JS"); // false s.startsWith("Hel"); // true s.endsWith("Script"); // true
查找：

slice / substring / substr

let s = "Hello JavaScript"; s.slice(6, 10); // "Java" 负索引支持 s.slice(-10); // "JavaScript" 后10个 s.substring(0, 5); // "Hello"

replace / split / join

let s = "Hello World"; s.replace("World", "JS"); // "Hello JS" (返回新字符串!) s.split(" "); // ["Hello","World"] s.split(""); // ["H","e","l","l","o",...]
