pop() 末尾删除  改变原数组

let arr = [1,2,3]; let last = arr.pop(); // last = 3, arr = [1,2]

shift() / unshift() 开头删/加  改变原数组

let arr = [1,2,3]; arr.unshift(0); // 开头插入 0 → [0,1,2,3] arr.shift(); // 删除开头元素

splice() 任意位置增删  改变原数组

// splice(起始索引, 删除数量, ...要插入的元素) let arr = [1,2,3,4,5]; arr.splice(1, 2, "a", "b"); // 从索引 1 开始删 2 个，再插入 "a","b"

sort() 排序  改变原数组

let arr = [3,1,4,1,5,9]; arr.sort(function(a, b){ return a - b; }); // 不传参数则按字符串排序！

② 返回新数组（Pure）— 不改变原数组

slice() 切片  返回新数组

let arr = [0,1,2,3,4]; arr.slice(1, 4); // [1,2,3] 含头不含尾 arr.slice(2); // [2,3,4] 从索引 2 到末尾 arr.slice(-2); // [3,4] 倒数两个

concat() 拼接  返回新数组

let arr = [1,2]; arr.concat([3,4], 5); // → [1,2,3,4,5]

filter() 过滤  返回新数组

let arr = [1,2,3,4,5]; arr.filter(function(n){ return n % 2 === 0; }); // → [2,4] 保留满足条件的元素

map() 映射  返回新数组

let arr = [1,2,3]; arr.map(function(n){ return n * 2; }); // → [2,4,6] 对每个元素做变换

reduce() 归约  返回单值

let arr = [1,2,3,4]; arr.reduce(function(acc, n){ return acc + n; }, 0); // 累加: 0+1=1 → 1+2=3 → 3+3=6 → 6+4=10

find() / some() / every() 查找检测  返回值

arr.find(n => n > 3); // 返回第一个 >3 的元素 arr.some(n => n % 2 === 0); // 是否存在偶数 → true/false arr.every(n => n > 0); // 是否全部 >0 → true/false arr.includes(3); // 是否包含 3 → true/false
