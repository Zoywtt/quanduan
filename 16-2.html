① 条件判断（Conditional）

if — else if — else 多分支

// 根据分数判定等级 function zyf_getGrade(score) { if (score >= 90) return "A"; else if (score >= 80) return "B"; else if (score >= 70) return "C"; else if (score >= 60) return "D"; else return "不及格"; }

switch 多分支

// 根据数字输出中文星期 function zyf_weekDay(n) { switch (n) { case 1: return "星期一"; case 2: return "星期二"; case 3: return "星期三"; case 4: return "星期四"; case 5: return "星期五"; case 6: return "星期六"; case 7: return "星期日"; default: return "❓ 非法输入"; } }

三元运算符 condition ? a : b

// 奇偶判断 —— 三元替代 if/else function zyf_parity(n) { return n % 2 === 0 ? "偶数 ✅" : "奇数 🔺"; }
② 循环（Loop）

for 循环 —— 固定次数

// 1 到 n 的累加和 function zyf_sumFor(n) { let sum = 0; for (let i = 1; i <= n; i++) { sum += i; } return sum; }

while 循环 —— 条件驱动

// 计算 2 的几次方刚好 ≥ 输入值 function zyf_powerOfTwo(target) { let result = 1, count = 0; while (result < target) { result *= 2; count++; } return "2^" + count + " = " + result; }

break 与 continue

// 输出 1-20 中，跳过 7 的倍数，遇到 15 停止 function zyf_breakContinue() { let arr = []; for (let i = 1; i <= 20; i++) { if (i === 15) break; // 遇到 15 就停止 if (i % 7 === 0) continue; // 跳过 7 的倍数 arr.push(i); } return arr; }

for...of —— 遍历数组

// 统计数组中每个字符串的长度 function zyf_forOf(arr) { let result = []; for (let item of arr) { result.push(item + "(" + item.length + ")"); } return result; }
